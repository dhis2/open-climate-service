// One feature's values over time, from the rows API. A feature with several values per // period (per class, reducer, weights or forecast) is charted one value at a time. (function () { var figure = document.getElementById("table-chart"); var featureSelect = document.getElementById("chart-feature"); var seriesField = document.getElementById("chart-series-field"); var seriesSelect = document.getElementById("chart-series"); var svg = figure.querySelector("svg"); var tooltip = figure.querySelector(".chart-tooltip"); var caption = document.getElementById("chart-caption"); var unit = figure.getAttribute("data-unit"); var SVG = "http://www.w3.org/2000/svg"; var series = {}; var points = []; // Each request's number, so a response to a superseded selection is dropped. var request = 0; function seriesLabel(row) { var parts = []; if (row["class"] != null) parts.push(row["class"]); if (row.reducer != null) parts.push(row.reducer); if (row.weights != null) parts.push("weighted by " + row.weights); if (row.member != null) parts.push("member " + row.member); if (row.reference_time != null) parts.push("issued " + row.reference_time.slice(0, 16).replace("T", " ")); return parts.join(", "); } function el(name, attrs, parent) { var node = document.createElementNS(SVG, name); Object.keys(attrs).forEach(function (key) { node.setAttribute(key, attrs[key]); }); parent.appendChild(node); return node; } function format(value) { return value == null ? "No value" : Number(value.toPrecision(6)).toLocaleString() + (unit ? " " + unit : ""); } function ticks(min, max, count) { // The smallest of 1, 2 or 5 times a power of ten that needs no more than *count* steps. var raw = (max - min) / count; var power = Math.pow(10, Math.floor(Math.log10(raw))); var step = [1, 2, 5, 10] .map(function (factor) { return factor * power; }) .filter(function (candidate) { return candidate >= raw; })[0]; var out = []; // From a round number at or below *min* to one at or above *max*, so the axis ends on a tick. for (var tick = Math.floor(min / step) * step; tick < max + step - step / 1e6; tick += step) out.push(tick); return out; } function draw() { svg.innerHTML = ""; tooltip.hidden = true; var width = Math.max(figure.clientWidth, 280); var height = 220; var margin = { top: 12, right: 12, bottom: 28, left: 56 }; svg.setAttribute("viewBox", "0 0 " + width + " " + height); svg.setAttribute("width", width); svg.setAttribute("height", height); var values = points.filter(function (point) { return point.value != null; }); if (!values.length) return; var low = Math.min.apply(null, values.map(function (point) { return point.value; })); var high = Math.max.apply(null, values.map(function (point) { return point.value; })); if (low === high) { low -= Math.abs(low) * 0.1 || 1; high += Math.abs(high) * 0.1 || 1; } var yTicks = ticks(low, high, 4); low = yTicks[0]; high = yTicks[yTicks.length - 1]; var first = points[0].time; var last = points[points.length - 1].time; function x(time) { return margin.left + (last === first ? 0.5 : (time - first) / (last - first)) * (width - margin.left - margin.right); } function y(value) { return height - margin.bottom - ((value - low) / (high - low)) * (height - margin.top - margin.bottom); } yTicks.forEach(function (tick) { el("line", { x1: margin.left, x2: width - margin.right, y1: y(tick), y2: y(tick), class: "grid" }, svg); el("text", { x: margin.left - 6, y: y(tick) + 4, "text-anchor": "end", class: "axis" }, svg).textContent = Number(tick.toPrecision(6)).toLocaleString(); }); var labelEvery = Math.ceil(points.length / Math.max(Math.floor((width - margin.left) / 72), 1)); points.forEach(function (point, index) { if (index % labelEvery !== 0) return; el("text", { x: x(point.time), y: height - 8, "text-anchor": "middle", class: "axis" }, svg).textContent = point.period; }); // A missing value breaks the line rather than being drawn as zero. var path = ""; var pen = false; points.forEach(function (point) { if (point.value == null) { pen = false; return; } path += (pen ? "L" : "M") + x(point.time).toFixed(1) + "," + y(point.value).toFixed(1); pen = true; }); el("path", { d: path, class: "line" }, svg); if (points.length <= 60) { values.forEach(function (point) { el("circle", { cx: x(point.time), cy: y(point.value), r: 4, class: "marker" }, svg); }); } var cross = el("line", { y1: margin.top, y2: height - margin.bottom, class: "crosshair", visibility: "hidden" }, svg); var focus = el("circle", { r: 5, class: "marker focus", visibility: "hidden" }, svg); var hit = el("rect", { x: margin.left, y: 0, width: width - margin.left - margin.right, height: height, fill: "transparent" }, svg); hit.addEventListener("pointermove", function (event) { var box = svg.getBoundingClientRect(); var at = ((event.clientX - box.left) / box.width) * width; var nearest = points.reduce(function (best, point) { return Math.abs(x(point.time) - at) < Math.abs(x(best.time) - at) ? point : best; }); cross.setAttribute("x1", x(nearest.time)); cross.setAttribute("x2", x(nearest.time)); cross.setAttribute("visibility", "visible"); focus.setAttribute("visibility", nearest.value == null ? "hidden" : "visible"); if (nearest.value != null) { focus.setAttribute("cx", x(nearest.time)); focus.setAttribute("cy", y(nearest.value)); } tooltip.textContent = nearest.period + ": " + format(nearest.value); tooltip.hidden = false; var left = (x(nearest.time) / width) * box.width; tooltip.style.left = Math.min(Math.max(left - tooltip.offsetWidth / 2, 0), box.width - tooltip.offsetWidth) + "px"; }); hit.addEventListener("pointerleave", function () { cross.setAttribute("visibility", "hidden"); focus.setAttribute("visibility", "hidden"); tooltip.hidden = true; }); } function show() { points = (series[seriesSelect.value] || []).slice().sort(function (a, b) { return a.time - b.time; }); var name = featureSelect.options[featureSelect.selectedIndex].text; var of = seriesSelect.value ? " (" + seriesSelect.value + ")" : ""; var missing = points.filter(function (point) { return point.value == null; }).length; caption.textContent = points.length ? name + of + ": " + points.length + " periods, " + points[0].period + " to " + points[points.length - 1].period + (missing ? ", " + missing + " without a value." : ".") : "No values for " + name + "."; svg.setAttribute("aria-label", "Values of " + name + of + " over time"); draw(); } function load() { caption.textContent = "Loading values…"; var current = ++request; var option = featureSelect.options[featureSelect.selectedIndex]; var url = figure.getAttribute("data-rows-url") + "?collection=" + encodeURIComponent(option.getAttribute("data-collection")) + "&feature=" + encodeURIComponent(featureSelect.value); fetch(url, { headers: { Accept: "application/json" } }) .then(function (response) { return response.ok ? response.json() : Promise.reject(response.status); }) .then(function (body) { if (current !== request) return; series = {}; body.items.forEach(function (row) { var key = seriesLabel(row); (series[key] = series[key] || []).push({ time: Date.parse(row.period_start), period: row.period, value: row.value, }); }); var keys = Object.keys(series).sort(); var previous = seriesSelect.value; seriesSelect.innerHTML = ""; keys.forEach(function (key) { var option = document.createElement("option"); option.value = key; option.textContent = key || "Value"; seriesSelect.appendChild(option); }); if (keys.indexOf(previous) >= 0) seriesSelect.value = previous; seriesField.hidden = keys.length < 2; show(); }) .catch(function () { if (current !== request) return; series = {}; points = []; svg.innerHTML = ""; caption.textContent = "Could not load the values of this feature."; }); } featureSelect.addEventListener("change", load); seriesSelect.addEventListener("change", show); window.addEventListener("resize", draw); load(); })();
